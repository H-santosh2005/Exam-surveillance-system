# Optional: enables the FaceNet-style /api/face/enroll and /api/face/verify
# endpoints (synopsis Sec 5.1). Requires dlib, which needs a C++ build
# toolchain (CMake + a compiler) to compile from source.
#
# Install AFTER requirements.txt, once your build tools are set up:
#   pip install -r requirements-face.txt
#
# Windows: the easiest path is usually Anaconda/Miniconda -
#   conda install -c conda-forge dlib
#   pip install face_recognition
# ...or install "Desktop development with C++" via the Visual Studio Build
# Tools installer, plus CMake (https://cmake.org/download/), then:
#   pip install cmake dlib face_recognition
#
# macOS:  brew install cmake && pip install dlib face_recognition
# Linux:  sudo apt install cmake build-essential && pip install dlib face_recognition
#
# Without this installed, every other route still works - /api/face/* will
# return a clear "face_recognition is not installed" error instead of
# crashing the server (see backend/utils/vision.py).
dlib==19.24.6
face_recognition==1.3.0
